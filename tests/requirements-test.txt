# Stage 1 test-runner dependencies.
#
# BOOTSTRAP (run once per machine, and again after the runner container is recreated):
#
#   docker exec -w /workspace evovis-runner-1 \
#     python -m pip install --no-deps --target /workspace/tests/.deps -r tests/requirements-test.txt
#
# RUN THE SUITE:
#
#   docker exec -w /workspace -e PYTHONPATH=/workspace/tests/.deps \
#     evovis-runner-1 python -m pytest tests
#
# WHY --target AND NOT THE CONDA ENVIRONMENT
# The `exp` environment is locked (docker/env-lock/, risk R19: only the editable
# `noisyvis` entry may ever differ from the lock). Installing pytest into `exp`
# would break that check, so the test runner lives in this bind-mounted directory
# instead. It therefore also survives the container recreations in Stages 3 and 4.
#
# THIS IS NOT A STANDALONE ENVIRONMENT -- READ BEFORE CHANGING
# pytest 8.4.2 requires, on CPython 3.11 / Linux:
#     iniconfig, packaging, pluggy, pygments
# (colorama, exceptiongroup and tomli are excluded by environment markers:
#  sys_platform == "win32" and python_version < "3.11").
# Only three of those are pinned below. `packaging` (25.0) and `pygments` (2.19.2)
# are taken from the locked `exp` environment ON PURPOSE: vendoring them here would
# shadow the locked versions via PYTHONPATH. Recreating this test environment thus
# requires `exp`, or equivalents of those two packages, to be present.
# tests/conftest.py asserts both are importable and meet pytest's minimums.
#
# --no-deps is safe precisely because the list below is complete by that analysis.

pytest==8.4.2
pluggy==1.6.0
iniconfig==2.3.0
